================================================================================
Postfix increment accepts calls, ref conditionals, null-forgiving values, and pointers
================================================================================

unsafe class PostfixOperands {
    static int value;
    static ref int F() => ref value;
    ref int M() => ref value;

    static void Cases(PostfixOperands obj, int[] arr, int i, byte* pointer,
        bool condition, ref int left, ref int right) {
        F()++;
        obj.M()++;
        arr[i]++;
        (condition ? ref left : ref right)++;
        left!++;
        (left!)++;
        *pointer++ = 1;
    }
}

struct Counter {
    public static Counter operator ++(Counter value) => value;
    public void IncrementThis() { this++; }
}
