==================
Conditional namespace aliases
==================
namespace N {
#if ONE
extern alias X;
#elif TWO
extern alias Y;
#else
extern alias Z;
#endif
using Y = X::A;
class C {}
}

---

(compilation_unit
  (namespace_declaration
    name: (identifier)
    body: (declaration_list
      (preproc_if
        condition: (identifier)
        (extern_alias_directive
          name: (identifier))
        alternative: (preproc_elif
          condition: (identifier)
          (extern_alias_directive
            name: (identifier))
          alternative: (preproc_else
            (extern_alias_directive
              name: (identifier)))))
      (using_directive
        name: (identifier)
        (alias_qualified_name
          alias: (identifier)
          name: (identifier)))
      (class_declaration
        name: (identifier)
        body: (declaration_list)))))
